# Phase 4: Main Menu & Game Setup (M147–M149) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire FA's main menu, skirmish lobby, and options dialog so the player can launch the game, configure a skirmish match against AI, and adjust settings.

**Architecture:** Phase 3 established the engine state machine (INIT→FRONT_END→LOADING→GAME→SCORE). Phase 4 implements the FRONT_END state — the main menu that FA's `main.lua:CreateUI()` builds, the skirmish lobby from `lobby.lua`, and the options dialog from `options.lua`. The engine already has stub lobby bindings (`InternalCreateLobby`, `lobby:HostGame`, etc.) and real file I/O (`DiskFindFiles`, `DiskGetFileInfo`, `exists`). Phase 4 upgrades lobby stubs, adds audio/movie globals, and implements the game launch sequence. No networking — single-player only.

**Tech Stack:** C++17, Lua 5.0, Vulkan, Catch2, CMake + vcpkg

**Spec:** [`docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md`](docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md)

/// soundTable = {Cue='UI_Menu_Click_01', Bank='Interface'} or string
static int l_PlaySound(lua_State* L) {
    auto* sim = get_sim(L);
    auto* snd = sim ? sim->sound_manager() : nullptr;

    if (!snd) { lua_pushnumber(L, 0); return 1; }

    std::string cue;
    if (lua_type(L, 1) == LUA_TSTRING) {
        cue = lua_tostring(L, 1);
    } else if (lua_istable(L, 1)) {
        lua_pushstring(L, "Cue");
        lua_rawget(L, 1);
        if (lua_type(L, -1) == LUA_TSTRING) cue = lua_tostring(L, -1);
        lua_pop(L, 1);
    }

    if (cue.empty()) { lua_pushnumber(L, 0); return 1; }

    // Play as 2D (non-positional) for UI sounds
    u32 handle = snd->play_ui_sound(cue);
    lua_pushnumber(L, handle);
    return 1;
}

}

```cpp
#pragma once

#include <string>
#include <unordered_map>

struct lua_State;

namespace osc {

/// Simple key→Lua-value store for cross-state communication.
/// Used by GetFrontEndData / SetFrontEndData.
/// Values are stored as Lua registry refs.
class FrontEndData {
public:
    void set(lua_State* L, const std::string& key, int value_idx);
    void get(lua_State* L, const std::string& key);
    void clear(lua_State* L);

private:
    std::unordered_map<std::string, int> refs_; // key → luaL_ref
};

} // namespace osc
```

- [ ] **Step 2: Write FrontEndData implementation inline**

Since this is small, implement in the header or a cpp:

```cpp
#include "core/front_end_data.hpp"

extern "C" {
#include <lua.h>
#include <lauxlib.h>
}

namespace osc {

void FrontEndData::set(lua_State* L, const std::string& key, int value_idx) {
    // Remove old ref if exists
    auto it = refs_.find(key);
    if (it != refs_.end()) {
        luaL_unref(L, LUA_REGISTRYINDEX, it->second);
    }
    lua_pushvalue(L, value_idx);
    refs_[key] = luaL_ref(L, LUA_REGISTRYINDEX);
}

void FrontEndData::get(lua_State* L, const std::string& key) {
    auto it = refs_.find(key);
    if (it != refs_.end()) {
        lua_rawgeti(L, LUA_REGISTRYINDEX, it->second);
    } else {
        lua_pushnil(L);
    }
}

void FrontEndData::clear(lua_State* L) {
    for (auto& [k, ref] : refs_) {
        luaL_unref(L, LUA_REGISTRYINDEX, ref);
    }
    refs_.clear();
}

} // namespace osc
```

```cpp
/// LaunchSinglePlayerSession(sessionConfig) — launch a game from lobby config.
/// sessionConfig table format:
///   {ScenarioFile = "/maps/.../scenario.lua",
///    Armies = {{name="ARMY_1", faction=1, ...}, {name="ARMY_2", ...}},
///    Options = {Victory="demoralization", ...}}
static int l_LaunchSinglePlayerSession(lua_State* L) {
    if (!lua_istable(L, 1)) {
        spdlog::warn("LaunchSinglePlayerSession: expected table arg");
        return 0;
    }

    // Read scenario file path
    lua_pushstring(L, "ScenarioFile");
    lua_rawget(L, 1);
    std::string scenario;
    if (lua_type(L, -1) == LUA_TSTRING) {
        scenario = lua_tostring(L, -1);
    }
    lua_pop(L, 1);

    if (scenario.empty()) {
        spdlog::warn("LaunchSinglePlayerSession: no ScenarioFile");
        return 0;
    }

    // Store the session config in FrontEndData so the loader can read it
    auto* fed = get_front_end_data(L);
    if (fed) fed->set(L, "sessionConfig", 1);

    // Store scenario path in registry for the game loader
    lua_pushstring(L, "__osc_launch_scenario");
    lua_pushstring(L, scenario.c_str());
    lua_rawset(L, LUA_REGISTRYINDEX);

    // Signal main.cpp to transition: FRONT_END → LOADING → GAME
    lua_pushstring(L, "__osc_launch_requested");
    lua_pushboolean(L, 1);
    lua_rawset(L, LUA_REGISTRYINDEX);

    spdlog::info("LaunchSinglePlayerSession: scenario={}", scenario);
    return 0;
}
```

/// Prefs.GetFromCurrentProfile(key [, default]) → value
static int l_GetFromCurrentProfile(lua_State* L) {
    int nargs = lua_gettop(L); // save arg count BEFORE pushing anything
    const char* key = luaL_checkstring(L, 1);
    std::string full_key = std::string("profile.") + key;

    // Delegate to GetPreference
    lua_pushstring(L, "GetPreference");
    lua_rawget(L, LUA_GLOBALSINDEX);
    if (lua_isfunction(L, -1)) {
        lua_pushstring(L, full_key.c_str());
        if (nargs >= 2) {
            lua_pushvalue(L, 2); // default value from original arg
        } else {
            lua_pushnil(L);
        }
        if (lua_pcall(L, 2, 1, 0) == 0) return 1;
        lua_pop(L, 1);
    } else {
        lua_pop(L, 1);
    }
    lua_pushnil(L);
    return 1;
}

/// Prefs.SetToCurrentProfile(key, val)
static int l_SetToCurrentProfile(lua_State* L) {
    const char* key = luaL_checkstring(L, 1);
    std::string full_key = std::string("profile.") + key;

    lua_pushstring(L, "SetPreference");
    lua_rawget(L, LUA_GLOBALSINDEX);
    if (lua_isfunction(L, -1)) {
        lua_pushstring(L, full_key.c_str());
        lua_pushvalue(L, 2);
        if (lua_pcall(L, 2, 0, 0) != 0) {
            lua_pop(L, 1);
        }
    } else {
        lua_pop(L, 1);
    }
    return 0;
}
