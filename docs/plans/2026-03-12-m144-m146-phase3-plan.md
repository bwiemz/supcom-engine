# Phase 3: Game Session Flow (M144–M146) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Formalize the engine state machine, implement per-frame Lua beat callbacks, and wire the score/game-over screen so a game session has a complete lifecycle.

**Architecture:** The current game loop in `main.cpp` uses ad-hoc booleans (`sim_paused`, `sim_speed`) with no formal state machine. Phase 3 introduces a `GameState` enum and transition logic, the beat function system that FA's Lua code depends on for per-frame updates, and score screen support. All bindings register on `ui_L` and bridge to `main.cpp`'s loop state via registry pointers.

**Tech Stack:** C++17, Lua 5.0, Vulkan, Catch2, CMake + vcpkg

**Spec:** [`docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md`](docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md)

**Prerequisites:** Phase 1 must be implemented first. This plan assumes `ui_bindings.cpp` exists (created by M135), dual Lua states are operational (`ui_lua_state` and `sim_lua_state` in `main.cpp`), and `get_sim(L)` helper exists in `ui_bindings.cpp` (fetches `SimState*` from registry via `osc_sim_state` key).

#pragma once

#include "core/types.hpp"
#include <string>

struct lua_State;

namespace osc {

/// Engine state machine: INIT → FRONT_END → LOADING → GAME → SCORE → FRONT_END
enum class GameState : u8 {
    INIT,       // Engine startup, loading core resources
    FRONT_END,  // Main menu (M147)
    LOADING,    // Map/session loading screen
    GAME,       // In-game (sim running, HUD active)
    SCORE,      // Post-game score screen
};

/// Returns the FA-expected string for GetCurrentUIState().
const char* game_state_to_string(GameState s);

/// Manages state transitions and fires Lua callbacks on each transition.
class GameStateManager {
public:
    GameState current() const { return state_; }

    /// Transition to a new state. Calls SetupUI() on ui_L before entering.
    /// Returns false if the transition is invalid.
    bool transition_to(GameState new_state, lua_State* ui_L);

    /// Pause/resume (only valid in GAME state)
    bool paused() const { return paused_; }
    void set_paused(bool p, lua_State* ui_L);

    /// Sim speed multiplier
    f64 speed() const { return speed_; }
    void set_speed(f64 s);

    /// Game-over flag (set when EndGame fires)
    bool game_over() const { return game_over_; }
    void set_game_over(bool v) { game_over_ = v; }

private:
    GameState state_ = GameState::INIT;
    bool paused_ = false;
    bool game_over_ = false;
    f64 speed_ = 1.0;

    /// Call SetupUI() on ui_L (cursor, prefs, skin init).
    static void call_setup_ui(lua_State* ui_L);
};

} // namespace osc
```

extern "C" {
#include <lua.h>

namespace osc {

const char* game_state_to_string(GameState s) {
    switch (s) {
    case GameState::INIT:      return "init";
    case GameState::FRONT_END: return "front-end";
    case GameState::LOADING:   return "loading";
    case GameState::GAME:      return "game";
    case GameState::SCORE:     return "score";
    }
    return "unknown";
}

bool GameStateManager::transition_to(GameState new_state, lua_State* ui_L) {
    if (new_state == state_) return true;

    spdlog::info("GameState: {} -> {}", game_state_to_string(state_),
                 game_state_to_string(new_state));

    // Call SetupUI() before entering the new state
    if (ui_L) call_setup_ui(ui_L);

    state_ = new_state;

    // Reset per-state flags
    if (new_state == GameState::GAME) {
        paused_ = false;
        game_over_ = false;
        speed_ = 1.0;
    }

    return true;
}

void GameStateManager::set_paused(bool p, lua_State* ui_L) {
    if (paused_ == p) return;
    paused_ = p;

    // Fire OnPause() or OnResume() into ui_L
    if (ui_L) {
        lua_pushstring(ui_L, p ? "OnPause" : "OnResume");
        lua_rawget(ui_L, LUA_GLOBALSINDEX);
        if (lua_isfunction(ui_L, -1)) {
            if (lua_pcall(ui_L, 0, 0, 0) != 0) {
                spdlog::warn("{} error: {}", p ? "OnPause" : "OnResume",
                             lua_tostring(ui_L, -1));
                lua_pop(ui_L, 1);
            }
        } else {
            lua_pop(ui_L, 1);
        }
    }
}

void GameStateManager::set_speed(f64 s) {
    if (s < 0.0) s = 0.0;
    if (s > 10.0) s = 10.0;
    speed_ = s;
}

void GameStateManager::call_setup_ui(lua_State* ui_L) {
    lua_pushstring(ui_L, "SetupUI");
    lua_rawget(ui_L, LUA_GLOBALSINDEX);
    if (lua_isfunction(ui_L, -1)) {
        if (lua_pcall(ui_L, 0, 0, 0) != 0) {
            spdlog::warn("SetupUI error: {}", lua_tostring(ui_L, -1));
            lua_pop(ui_L, 1);
        }
    } else {
        lua_pop(ui_L, 1);
    }
}

} // namespace osc

#pragma once

#include <string>
#include <vector>

struct lua_State;

namespace osc::lua {

/// Stores Lua function references registered via AddBeatFunction.
/// Functions are called once per frame in registration order.
class BeatFunctionRegistry {
public:
    /// Register a beat function. If `name` is provided, it can be removed by name.
    /// The Lua function at stack position `func_idx` is stored via luaL_ref.
    void add(lua_State* L, int func_idx, const std::string& name = "");

    /// Remove a beat function by its Lua function reference.
    /// Matches by comparing the Lua function value.
    void remove(lua_State* L, int func_idx);

    /// Remove a beat function by name. Calls luaL_unref to free registry refs.
    void remove_by_name(const std::string& name, lua_State* L);

    /// Call all registered beat functions. Called once per frame.
    void fire_all(lua_State* L);

    /// Clear all registered functions (e.g., on state transition).
    void clear(lua_State* L);

    size_t count() const { return entries_.size(); }

private:
    struct Entry {
        int lua_ref = -2; // LUA_NOREF
        std::string name;
    };
    std::vector<Entry> entries_;
};

} // namespace osc::lua
```

- [ ] **Step 2: Write beat system implementation**

```cpp
#include "lua/beat_system.hpp"
#include <spdlog/spdlog.h>
#include <algorithm>

extern "C" {
#include <lua.h>
#include <lauxlib.h>
}

namespace osc::lua {

void BeatFunctionRegistry::add(lua_State* L, int func_idx, const std::string& name) {
    lua_pushvalue(L, func_idx); // copy function to top
    int ref = luaL_ref(L, LUA_REGISTRYINDEX);
    entries_.push_back({ref, name});
    spdlog::debug("BeatFunction added: ref={} name='{}'", ref, name);
}

void BeatFunctionRegistry::remove(lua_State* L, int func_idx) {
    // Compare the Lua function value against stored refs
    for (auto it = entries_.begin(); it != entries_.end(); ++it) {
        lua_rawgeti(L, LUA_REGISTRYINDEX, it->lua_ref);
        lua_pushvalue(L, func_idx);
        bool equal = lua_equal(L, -1, -2) != 0;
        lua_pop(L, 2);
        if (equal) {
            luaL_unref(L, LUA_REGISTRYINDEX, it->lua_ref);
            entries_.erase(it);
            return;
        }
    }
}

void BeatFunctionRegistry::remove_by_name(const std::string& name, lua_State* L) {
    for (auto it = entries_.begin(); it != entries_.end(); ) {
        if (it->name == name) {
            luaL_unref(L, LUA_REGISTRYINDEX, it->lua_ref);
            it = entries_.erase(it);
        } else {
            ++it;
        }
    }
}

void BeatFunctionRegistry::fire_all(lua_State* L) {
    // Snapshot refs to avoid iterator invalidation if callbacks add/remove entries
    std::vector<std::pair<int, std::string>> snapshot;
    snapshot.reserve(entries_.size());
    for (const auto& e : entries_) {
        snapshot.emplace_back(e.lua_ref, e.name);
    }

    for (const auto& [ref, name] : snapshot) {
        lua_rawgeti(L, LUA_REGISTRYINDEX, ref);
        if (lua_isfunction(L, -1)) {
            if (lua_pcall(L, 0, 0, 0) != 0) {
                spdlog::warn("BeatFunction '{}' error: {}",
                             name, lua_tostring(L, -1));
                lua_pop(L, 1);
            }
        } else {
            lua_pop(L, 1);
        }
    }
}

void BeatFunctionRegistry::clear(lua_State* L) {
    for (auto& e : entries_) {
        luaL_unref(L, LUA_REGISTRYINDEX, e.lua_ref);
    }
    entries_.clear();
}

} // namespace osc::lua
```

/// AddBeatFunction(func, name) — register a per-frame callback.
/// Called by FA's UI code to register update functions.
static int l_AddBeatFunction(lua_State* L) {
    auto* reg = get_beat_registry(L);
    if (!reg || !lua_isfunction(L, 1)) return 0;

    std::string name;
    if (lua_type(L, 2) == LUA_TSTRING) {
        name = lua_tostring(L, 2);
    }

    reg->add(L, 1, name);
    return 0;
}

/// RemoveBeatFunction(func) — unregister a per-frame callback.
static int l_RemoveBeatFunction(lua_State* L) {
    auto* reg = get_beat_registry(L);
    if (!reg) return 0;

    if (lua_isfunction(L, 1)) {
        reg->remove(L, 1);
    } else if (lua_type(L, 1) == LUA_TSTRING) {
        reg->remove_by_name(lua_tostring(L, 1), L);
    }
    return 0;
}

/// SessionGetScenarioInfo() → table with scenario metadata
/// Reads from SimState's cached scenario info (stored during session boot).
static int l_ui_SessionGetScenarioInfo(lua_State* L) {
    auto* sim = get_sim(L);
    if (!sim) { lua_newtable(L); return 1; }

    lua_newtable(L);
    auto set_str = [&](const char* k, const char* v) {
        lua_pushstring(L, k); lua_pushstring(L, v); lua_rawset(L, -3);
    };
    auto set_num = [&](const char* k, f64 v) {
        lua_pushstring(L, k); lua_pushnumber(L, v); lua_rawset(L, -3);
    };

    // Read from scenario info stored in registry during session boot
    lua_pushstring(L, "__osc_scenario_path");
    lua_rawget(L, LUA_REGISTRYINDEX);
    const char* path = lua_isnil(L, -1) ? "" : lua_tostring(L, -1);
    lua_pop(L, 1);

    set_str("name", path);
    set_str("file", path);

    // Map size from terrain
    auto* terrain = sim->terrain();
    if (terrain) {
        set_num("size_x", terrain->width());
        set_num("size_z", terrain->height());
    }

    // Armies subtable
    lua_pushstring(L, "Armies");
    lua_newtable(L);
    for (size_t i = 0; i < sim->army_count(); ++i) {
        auto* brain = sim->army_at(i);
        if (brain) {
            lua_newtable(L);
            lua_pushstring(L, "name");
            lua_pushstring(L, brain->name().c_str());
            lua_rawset(L, -3);
            lua_rawseti(L, -2, static_cast<int>(i + 1));
        }
    }
    lua_rawset(L, -3);

    return 1;
}

/// GetArmyScore(armyIndex) → table with army stats for score screen.
/// Format: {general={score, currentunits, currentcap},
///          resources={massin={rate}, massout={rate}, energyin={rate}, energyout={rate},
///                     storage={maxMass, storedMass, maxEnergy, storedEnergy}},
///          Defeated=bool}
static int l_GetArmyScore(lua_State* L) {
    auto* sim = get_sim(L);
    int army_idx = static_cast<int>(luaL_checknumber(L, 1));

    lua_newtable(L); // root
    if (!sim) return 1;

    auto* brain = sim->get_army(army_idx);
    if (!brain) return 1;

    auto set_num = [&](const char* k, f64 v) {
        lua_pushstring(L, k);
        lua_pushnumber(L, v);
        lua_rawset(L, -3);
    };

    // general subtable
    lua_pushstring(L, "general");
    lua_newtable(L);
    set_num("score", brain->score());
    // Count live units for this army
    int unit_count = 0;
    auto& reg = sim->entity_registry();
    for (u32 id : reg.all_ids()) {
        auto* e = reg.get(id);
        if (e && e->is_unit() && e->army() == army_idx && !e->is_dead())
            ++unit_count;
    }
    set_num("currentunits", unit_count);
    set_num("currentcap", 1000); // unit cap (simplified)
    lua_rawset(L, -3); // set general

    // resources subtable
    lua_pushstring(L, "resources");
    lua_newtable(L);

    auto push_rate_table = [&](const char* name, f64 rate) {
        lua_pushstring(L, name);
        lua_newtable(L);
        lua_pushstring(L, "rate");
        lua_pushnumber(L, rate);
        lua_rawset(L, -3);
        lua_rawset(L, -3);
    };

    push_rate_table("massin", brain->income_mass());
    push_rate_table("massout", brain->expense_mass());
    push_rate_table("energyin", brain->income_energy());
    push_rate_table("energyout", brain->expense_energy());

    // storage subtable
    lua_pushstring(L, "storage");
    lua_newtable(L);
    set_num("maxMass", brain->max_mass());
    set_num("storedMass", brain->stored_mass());
    set_num("maxEnergy", brain->max_energy());
    set_num("storedEnergy", brain->stored_energy());
    lua_rawset(L, -3); // set storage
    lua_rawset(L, -3); // set resources

    // Defeated flag
    lua_pushstring(L, "Defeated");
    lua_pushboolean(L, brain->is_defeated() ? 1 : 0);
    lua_rawset(L, -3);

    return 1;
}
