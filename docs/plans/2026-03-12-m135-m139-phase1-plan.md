# Phase 1: WorldView & Command Infrastructure (M135–M139) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the foundation that all in-game FA Lua UI panels depend on — dual Lua VMs, game entry points, WorldView control, selection bridge, command mode, and build placement.

**Architecture:** The engine currently uses a single `lua_State*` for everything. Phase 1 splits this into `sim_L` (deterministic simulation) and `ui_L` (variable-rate UI). A minimal state machine boots into GAME state, calling FA's `StartGameUI()` → `CreateWldUIProvider()` → `CreateGameInterface()`. WorldView becomes a real UIControl subclass with camera projection and terrain raycasting. Selection, commands, and build placement bridge UI clicks to sim-side actions via `SimCallback` serialization.

**Tech Stack:** C++17, Lua 5.0, Vulkan, Catch2, CMake + vcpkg

**Spec:** [`docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md`](docs/plans/2026-03-12-m135-m152-skirmish-roadmap-design.md)

```cpp
#include "core/preferences.hpp"
#include <fstream>
#include <nlohmann/json.hpp>
#include <spdlog/spdlog.h>

namespace osc::core {

void Preferences::load(const std::string& path) {
    std::ifstream f(path);
    if (!f.is_open()) {
        spdlog::debug("Preferences: no file at {}", path);
        return;
    }
    try {
        nlohmann::json j;
        f >> j;
        // Flatten nested JSON into dot-notation keys
        std::function<void(const nlohmann::json&, const std::string&)> flatten;
        flatten = [&](const nlohmann::json& obj, const std::string& prefix) {
            for (auto& [k, v] : obj.items()) {
                auto full_key = prefix.empty() ? k : prefix + "." + k;
                if (v.is_object()) {
                    flatten(v, full_key);
                } else if (v.is_string()) {
                    data_[full_key] = v.get<std::string>();
                } else if (v.is_boolean()) {
                    data_[full_key] = v.get<bool>();
                } else if (v.is_number_float()) {
                    data_[full_key] = v.get<float>();
                } else if (v.is_number_integer()) {
                    data_[full_key] = v.get<int>();
                }
            }
        };
        flatten(j, "");
        spdlog::info("Preferences: loaded {} keys from {}", data_.size(), path);
    } catch (const std::exception& e) {
        spdlog::warn("Preferences: failed to parse {}: {}", path, e.what());
    }
}

void Preferences::save(const std::string& path) const {
    nlohmann::json root;
    for (auto& [key, val] : data_) {
        // Split dot-notation into nested JSON
        auto* node = &root;
        size_t start = 0;
        while (true) {
            size_t dot = key.find('.', start);
            if (dot == std::string::npos) {
                auto leaf = key.substr(start);
                std::visit([&](auto&& v) { (*node)[leaf] = v; }, val);
                break;
            }
            node = &((*node)[key.substr(start, dot - start)]);
            start = dot + 1;
        }
    }
    std::ofstream f(path);
    if (f.is_open()) {
        f << root.dump(2);
        spdlog::debug("Preferences: saved {} keys to {}", data_.size(), path);
    }
}

std::string Preferences::get_string(const std::string& key,
                                     const std::string& def) const {
    auto it = data_.find(key);
    if (it == data_.end()) return def;
    if (auto* s = std::get_if<std::string>(&it->second)) return *s;
    return def;
}

float Preferences::get_float(const std::string& key, float def) const {
    auto it = data_.find(key);
    if (it == data_.end()) return def;
    if (auto* f = std::get_if<float>(&it->second)) return *f;
    if (auto* i = std::get_if<int>(&it->second)) return static_cast<float>(*i);
    return def;
}

bool Preferences::get_bool(const std::string& key, bool def) const {
    auto it = data_.find(key);
    if (it == data_.end()) return def;
    if (auto* b = std::get_if<bool>(&it->second)) return *b;
    return def;
}

int Preferences::get_int(const std::string& key, int def) const {
    auto it = data_.find(key);
    if (it == data_.end()) return def;
    if (auto* i = std::get_if<int>(&it->second)) return *i;
    if (auto* f = std::get_if<float>(&it->second)) return static_cast<int>(*f);
    return def;
}

void Preferences::set_string(const std::string& key, const std::string& val) {
    data_[key] = val;
}

void Preferences::set_float(const std::string& key, float val) {
    data_[key] = val;
}

void Preferences::set_bool(const std::string& key, bool val) {
    data_[key] = val;
}

void Preferences::set_int(const std::string& key, int val) {
    data_[key] = val;
}

} // namespace osc::core
```

```cpp
#pragma once

#include <string>
#include <unordered_map>
#include <vector>

struct lua_State;

namespace osc::vfs {
class VirtualFileSystem;
}

namespace osc::core {

/// Caches localization strings from /loc/us/strings_db.lua.
/// LOC(key) does a single hash lookup — no Lua pcall per invocation.
/// FA convention: if key not found, return the key itself.
class Localization {
public:
    /// Add a single key→string mapping.
    void add(const std::string& key, const std::string& value);

    /// Load all strings from /loc/us/strings_db.lua via VFS + Lua.
    /// Parses the Lua file which sets globals like:
    ///   loc_key_001="value"
    /// Iterates the global table to extract all string mappings.
    void load_from_vfs(lua_State* L, vfs::VirtualFileSystem* vfs);

    /// Look up a localization key. Returns the value, or key itself if missing.
    /// Handles FA's "<LOC key>" wrapper format.
    const std::string& lookup(const std::string& key) const;

    /// LOCF: look up key, then substitute %s/%d args positionally.
    std::string format(const std::string& key,
                       const std::vector<std::string>& args) const;

    size_t size() const { return strings_.size(); }

private:
    std::unordered_map<std::string, std::string> strings_;
    mutable std::string fallback_; // for returning ref to missing keys
};

} // namespace osc::core
```

- [ ] **Step 4: Write Localization implementation**

Create `src/core/localization.cpp`:

```cpp
#include "core/localization.hpp"
#include "vfs/virtual_file_system.hpp"
#include <spdlog/spdlog.h>
#include <cstdio>

extern "C" {
#include <lua.h>
#include <lauxlib.h>
}

namespace osc::core {

void Localization::add(const std::string& key, const std::string& value) {
    strings_[key] = value;
}

void Localization::load_from_vfs(lua_State* L, vfs::VirtualFileSystem* vfs) {
    // Load strings_db.lua which sets global variables like:
    //   loc_key = "translated string"
    auto data = vfs->read_file("/loc/us/strings_db.lua");
    if (data.empty()) {
        spdlog::warn("Localization: /loc/us/strings_db.lua not found or empty");
        return;
    }

    // Execute in a temporary Lua state to avoid polluting globals
    lua_State* tmp = lua_open();
    luaopen_base(tmp);
    luaopen_string(tmp);

    if (luaL_loadbuffer(tmp, data.data(), data.size(), "strings_db") != 0 ||
        lua_pcall(tmp, 0, 0, 0) != 0) {
        spdlog::warn("Localization: failed to execute strings_db.lua: {}",
                     lua_tostring(tmp, -1));
        lua_close(tmp);
        return;
    }

    // Iterate all globals — strings_db sets flat globals
    lua_pushvalue(tmp, LUA_GLOBALSINDEX);
    lua_pushnil(tmp);
    while (lua_next(tmp, -2) != 0) {
        if (lua_type(tmp, -2) == LUA_TSTRING &&
            lua_type(tmp, -1) == LUA_TSTRING) {
            const char* k = lua_tostring(tmp, -2);
            const char* v = lua_tostring(tmp, -1);
            if (k && v) {
                strings_[k] = v;
                // Also store with <LOC key> wrapper
                strings_[std::string("<LOC ") + k + ">"] = v;
            }
        }
        lua_pop(tmp, 1); // pop value, keep key
    }
    lua_pop(tmp, 1); // pop globals table

    lua_close(tmp);
    spdlog::info("Localization: loaded {} strings", strings_.size());
}

const std::string& Localization::lookup(const std::string& key) const {
    auto it = strings_.find(key);
    if (it != strings_.end()) return it->second;

    // Try stripping <LOC ...> wrapper
    if (key.size() > 6 && key.substr(0, 5) == "<LOC " && key.back() == '>') {
        auto inner = key.substr(5, key.size() - 6);
        auto it2 = strings_.find(inner);
        if (it2 != strings_.end()) return it2->second;
    }

    // FA convention: return key itself if not found
    fallback_ = key;
    return fallback_;
}

std::string Localization::format(const std::string& key,
                                  const std::vector<std::string>& args) const {
    const auto& fmt = lookup(key);
    if (args.empty()) return fmt;

    // Simple positional substitution: replace %s and %d with args in order
    std::string result;
    result.reserve(fmt.size() + 64);
    size_t arg_idx = 0;
    for (size_t i = 0; i < fmt.size(); ++i) {
        if (fmt[i] == '%' && i + 1 < fmt.size() && arg_idx < args.size()) {
            char spec = fmt[i + 1];
            if (spec == 's' || spec == 'd' || spec == 'f') {
                result += args[arg_idx++];
                ++i; // skip specifier
                continue;
            }
        }
        result += fmt[i];
    }
    return result;
}

} // namespace osc::core
```

- [ ] **Step 2: Modify main.cpp to create dual Lua states**

The current flow in `main()` is:
1. Create `LuaState state`
2. Init + VFS mount
3. Blueprint loading
4. `SimState sim_state(state.raw(), &store)`
5. Boot sim (registers sim_bindings on `state`)

7. Session start
8. Game loop

New flow:
1. Create `LuaState sim_lua_state` (for simulation)
2. Init + VFS mount using `sim_lua_state`
3. Blueprint loading using `sim_lua_state`
4. `SimState sim_state(sim_lua_state.raw(), &store)`
5. Boot sim (registers sim_bindings on `sim_lua_state`)
6. Create `LuaState ui_lua_state` (for UI)
7. Share VFS + blueprint store pointers: `ui_lua_state.set_vfs(&vfs); ui_lua_state.set_blueprint_store(&store);`
8. Run init sequence on `ui_lua_state` (string polyfills, repr, config, utils, class, import)
9. Register moho class tables on `ui_lua_state` (for UI controls)
10. Register UI bindings on `ui_lua_state`
11. Session start (uses `sim_lua_state` for sim, `ui_lua_state` for UI callbacks)
12. Game loop passes both `sim_L` and `ui_L` where needed

```cpp
/// GetPreference(key, default) → value
/// FA calls this constantly for UI layout preferences.
static int l_GetPreference(lua_State* L) {
    const char* key = luaL_checkstring(L, 1);
    auto* prefs = get_prefs(L);
    if (!prefs) {
        // No prefs — return the default (arg 2)
        if (lua_gettop(L) >= 2) lua_pushvalue(L, 2);
        else lua_pushnil(L);
        return 1;
    }

    // Try to match type of default argument
    if (lua_gettop(L) >= 2) {
        int t = lua_type(L, 2);
        if (t == LUA_TSTRING) {
            auto val = prefs->get_string(key, lua_tostring(L, 2));
            lua_pushstring(L, val.c_str());
        } else if (t == LUA_TBOOLEAN) {
            bool val = prefs->get_bool(key, lua_toboolean(L, 2) != 0);
            lua_pushboolean(L, val ? 1 : 0);
        } else if (t == LUA_TNUMBER) {
            float val = prefs->get_float(key, static_cast<float>(lua_tonumber(L, 2)));
            lua_pushnumber(L, val);
        } else {
            // Table or other — return default as-is
            lua_pushvalue(L, 2);
        }
    } else {
        auto val = prefs->get_string(key, "");
        if (val.empty()) lua_pushnil(L);
        else lua_pushstring(L, val.c_str());
    }
    return 1;
}

/// SetPreference(key, value) — persists on next save
static int l_SetPreference(lua_State* L) {
    const char* key = luaL_checkstring(L, 1);

    else if (t == LUA_TNUMBER)
        prefs->set_float(key, static_cast<float>(lua_tonumber(L, 2)));
```

```cpp
/// WaitSeconds(n): convert seconds to frame count, set wait_until_tick.
/// ThreadManager::resume_all() is called with ui_frame_count each frame.
/// ThreadEntry.wait_until_tick is repurposed as "wait_until_frame" in UI context.
static int l_ui_WaitSeconds(lua_State* L) {
    auto* tm = get_ui_threads(L);
    auto* fc = get_ui_frame_count(L);
    if (!tm || !fc) return 0;
    f64 seconds = luaL_checknumber(L, 1);
    // Convert seconds to frames at ~60 FPS (coarse but functional).
    // FA uses WaitSeconds primarily for short delays (0.1-1.0s).
    u32 frames = static_cast<u32>(seconds * 60.0 + 0.5);
    if (frames < 1) frames = 1;

    // Find the current thread in ThreadManager and set its wait_until_tick
    // to current_frame + frames. This reuses ThreadManager's tick-based
    // scheduling by treating frames as ticks.
    // The coroutine yields; ThreadManager will resume it when
    // ui_frame_count >= wait_until_tick.
    // ThreadManager handles the yield internally via fork_thread's coroutine.

    // Approach: push the wait target into a Lua global that the thread
    // wrapper can read, then yield.
    lua_pushnumber(L, static_cast<lua_Number>(*fc + frames));
    return lua_yield(L, 1); // yield with 1 result (the target frame)
}

/// WaitTicks(n): in UI context, 1 tick = 1 frame.
static int l_ui_WaitTicks(lua_State* L) {
    auto* tm = get_ui_threads(L);
    auto* fc = get_ui_frame_count(L);
    if (!tm || !fc) return 0;
    int ticks = static_cast<int>(luaL_checknumber(L, 1));
    if (ticks < 1) ticks = 1;
    lua_pushnumber(L, static_cast<lua_Number>(*fc + ticks));
    return lua_yield(L, 1);
}
```

**Important Lua 5.0 note:** `lua_yield` returns `0` on success in Lua 5.0. The `lua_resume` return value is `0` for both completion and yield — check `lua_status()` to distinguish. The existing `ThreadManager::resume_all` handles this correctly.

**Alternative simpler approach:** If the yield-with-result approach is too complex, simply set `wait_until_tick` directly on the ThreadEntry before yielding. This requires exposing a method on ThreadManager to set the wait target for the current coroutine. Add to `ThreadManager`:

```cpp
/// Set the wait target (in tick/frame units) for the most recently
/// forked or currently running thread.
void set_current_wait(lua_State* co, u32 target_tick);
```

```cpp
bool WorldView::project(f32 wx, f32 wy, f32 wz,
                         f32& sx, f32& sy) const {
    if (!camera_ || vp_width_ == 0) return false;

    // Camera::view_proj(aspect) returns column-major 4x4
    f32 aspect = static_cast<f32>(vp_width_) / static_cast<f32>(vp_height_);
    auto vp = camera_->view_proj(aspect);

    // Transform world → clip space
    f32 cx = vp[0]*wx + vp[4]*wy + vp[8]*wz  + vp[12];
    f32 cy = vp[1]*wx + vp[5]*wy + vp[9]*wz  + vp[13];
    f32 cw = vp[3]*wx + vp[7]*wy + vp[11]*wz + vp[15];

    if (cw <= 0.001f) return false; // behind camera

    // Clip → NDC
    f32 ndx = cx / cw;
    f32 ndy = cy / cw;

    // NDC → screen (Vulkan: Y is top-down already)
    sx = (ndx * 0.5f + 0.5f) * static_cast<f32>(vp_width_);
    sy = (ndy * 0.5f + 0.5f) * static_cast<f32>(vp_height_);
    return true;
}

bool WorldView::get_mouse_world_pos(f32 sx, f32 sy,
                                      f32& wx, f32& wy, f32& wz) const {
    if (!camera_ || !terrain_ || vp_width_ == 0) return false;

    // If quadtree is available, use Camera::screen_to_world for flat-plane
    // approximation first, then refine with quadtree raycasting.
    // Camera already has screen_to_world(sx, sy, w, h, ground_y, &ox, &oz)
    // which handles the inverse VP math internally.
    if (terrain_quadtree_ && terrain_quadtree_->built()) {
        // Compute ray from screen point using camera position + direction
        f32 eye_x, eye_y, eye_z;
        camera_->eye_position(eye_x, eye_y, eye_z);

        // Get approximate ground hit for ray direction
        f32 ground_x, ground_z;
        f32 ground_y = 0.0f; // approximate
        if (!camera_->screen_to_world(sx, sy,
                                        static_cast<f32>(vp_width_),
                                        static_cast<f32>(vp_height_),
                                        ground_y, ground_x, ground_z)) {
            return false;
        }

        // Ray: origin=eye, direction=normalize(ground_hit - eye)
        f32 dx = ground_x - eye_x;
        f32 dy = ground_y - eye_y;
        f32 dz = ground_z - eye_z;
        f32 len = std::sqrt(dx*dx + dy*dy + dz*dz);
        if (len < 1e-8f) return false;
        dx /= len; dy /= len; dz /= len;

        return terrain_quadtree_->raycast(eye_x, eye_y, eye_z,
                                            dx, dy, dz, wx, wy, wz);
    }

    // Fallback: use Camera::screen_to_world with terrain height refinement
    f32 ground_x, ground_z;
    if (!camera_->screen_to_world(sx, sy,
                                    static_cast<f32>(vp_width_),
                                    static_cast<f32>(vp_height_),
                                    0.0f, ground_x, ground_z)) {
        return false;
    }
    wx = ground_x;
    wz = ground_z;
    wy = terrain_->get_height(wx, wz);
    return true;
}

void WorldView::zoom_scale(f32 x, f32 y, f32 rotation, f32 delta) {
    if (!camera_ || input_locked_) return;
    camera_->zoom(delta);
}
```

```cpp
#pragma once

#include "core/types.hpp"
#include <vector>

namespace osc::map {

/// Hierarchical heightmap for O(log N) screen ray → terrain intersection.
/// Each node stores the min/max height for its bounding region.
/// Rays that can't intersect a node's height range skip its subtree.
class TerrainQuadtree {
public:
    /// Build from heightmap. cell_size = world units per heightmap cell.
    void build(const f32* heights, u32 width, u32 height, f32 cell_size);

    /// Cast a ray and find intersection point.
    /// Returns true if hit, sets hx/hy/hz to intersection.
    bool raycast(f32 ox, f32 oy, f32 oz,
                 f32 dx, f32 dy, f32 dz,
                 f32& hx, f32& hy, f32& hz) const;

    bool built() const { return !nodes_.empty(); }

private:
    struct Node {
        f32 min_h, max_h;       // height range in this cell
        f32 x0, z0, x1, z1;    // world-space bounds
        i32 children[4] = {-1, -1, -1, -1}; // indices or -1 for leaf
    };

    void build_recursive(i32 node_idx, const f32* heights,
                         u32 hmap_w, u32 hmap_h, f32 cell_size,
                         u32 cx0, u32 cz0, u32 cx1, u32 cz1);

    bool raycast_node(i32 idx, f32 ox, f32 oy, f32 oz,
                      f32 dx, f32 dy, f32 dz,
                      f32& best_t, f32& hx, f32& hy, f32& hz) const;

    /// Ray-AABB intersection test (returns t_enter, t_exit).
    static bool ray_aabb(f32 ox, f32 oy, f32 oz,
                         f32 dx, f32 dy, f32 dz,
                         f32 x0, f32 y0, f32 z0,
                         f32 x1, f32 y1, f32 z1,
                         f32& t_enter, f32& t_exit);

    std::vector<Node> nodes_;
    const f32* heights_ = nullptr;
    u32 hmap_w_ = 0, hmap_h_ = 0;
    f32 cell_size_ = 1.0f;
};

} // namespace osc::map
```

```cpp
#include "map/terrain_quadtree.hpp"
#include <algorithm>
#include <cmath>
#include <limits>

namespace osc::map {

bool TerrainQuadtree::ray_aabb(f32 ox, f32 oy, f32 oz,
                                 f32 dx, f32 dy, f32 dz,
                                 f32 x0, f32 y0, f32 z0,
                                 f32 x1, f32 y1, f32 z1,
                                 f32& t_enter, f32& t_exit) {
    f32 tmin = -1e30f, tmax = 1e30f;
    auto slab = [&](f32 o, f32 d, f32 lo, f32 hi) {
        if (std::abs(d) < 1e-8f) {
            if (o < lo || o > hi) { tmin = 1e30f; return; }
        } else {
            f32 t1 = (lo - o) / d, t2 = (hi - o) / d;
            if (t1 > t2) std::swap(t1, t2);
            tmin = std::max(tmin, t1);
            tmax = std::min(tmax, t2);
        }
    };
    slab(ox, dx, x0, x1);
    slab(oy, dy, y0, y1);
    slab(oz, dz, z0, z1);
    if (tmin > tmax || tmax < 0) return false;
    t_enter = std::max(tmin, 0.0f);
    t_exit = tmax;
    return true;
}

void TerrainQuadtree::build(const f32* heights, u32 w, u32 h, f32 cell_size) {
    heights_ = heights;
    hmap_w_ = w;
    hmap_h_ = h;
    cell_size_ = cell_size;
    nodes_.clear();
    nodes_.reserve(w * h); // upper bound estimate
    // Root node covers entire heightmap
    nodes_.push_back({});
    build_recursive(0, heights, w, h, cell_size, 0, 0, w, h);
}

void TerrainQuadtree::build_recursive(i32 idx, const f32* heights,
                                        u32 w, u32 h, f32 cs,
                                        u32 cx0, u32 cz0, u32 cx1, u32 cz1) {
    auto& node = nodes_[idx];
    node.x0 = cx0 * cs;
    node.z0 = cz0 * cs;
    node.x1 = cx1 * cs;
    node.z1 = cz1 * cs;

    // Compute min/max height in this cell range
    node.min_h = std::numeric_limits<f32>::max();
    node.max_h = std::numeric_limits<f32>::lowest();
    for (u32 z = cz0; z < cz1 && z < h; ++z) {
        for (u32 x = cx0; x < cx1 && x < w; ++x) {
            f32 val = heights[z * w + x];
            node.min_h = std::min(node.min_h, val);
            node.max_h = std::max(node.max_h, val);
        }
    }

    // Leaf if region is small enough (4x4 or less)
    u32 dx = cx1 - cx0, dz = cz1 - cz0;
    if (dx <= 4 && dz <= 4) return;

    // Subdivide into 4 quadrants
    u32 mx = cx0 + dx / 2, mz = cz0 + dz / 2;
    u32 ranges[4][4] = {
        {cx0, cz0, mx, mz}, {mx, cz0, cx1, mz},
        {cx0, mz, mx, cz1}, {mx, mz, cx1, cz1}
    };
    for (int c = 0; c < 4; ++c) {
        if (ranges[c][0] >= ranges[c][2] || ranges[c][1] >= ranges[c][3]) continue;
        i32 child_idx = static_cast<i32>(nodes_.size());
        nodes_[idx].children[c] = child_idx;
        nodes_.push_back({});
        build_recursive(child_idx, heights, w, h, cs,
                         ranges[c][0], ranges[c][1], ranges[c][2], ranges[c][3]);
    }
}

bool TerrainQuadtree::raycast(f32 ox, f32 oy, f32 oz,
                                f32 dx, f32 dy, f32 dz,
                                f32& hx, f32& hy, f32& hz) const {
    if (nodes_.empty()) return false;
    f32 best_t = 1e30f;
    return raycast_node(0, ox, oy, oz, dx, dy, dz, best_t, hx, hy, hz);
}

bool TerrainQuadtree::raycast_node(i32 idx, f32 ox, f32 oy, f32 oz,
                                     f32 dx, f32 dy, f32 dz,
                                     f32& best_t,
                                     f32& hx, f32& hy, f32& hz) const {
    const auto& n = nodes_[idx];
    f32 t_enter, t_exit;
    if (!ray_aabb(ox, oy, oz, dx, dy, dz,
                  n.x0, n.min_h, n.z0, n.x1, n.max_h, n.z1,
                  t_enter, t_exit)) {
        return false;
    }
    if (t_enter >= best_t) return false; // already have a closer hit

    // Check if this is a leaf
    bool is_leaf = true;
    for (int c = 0; c < 4; ++c) {
        if (n.children[c] >= 0) { is_leaf = false; break; }
    }

    if (is_leaf) {
        // Linear search within this small cell for exact intersection
        f32 step = cell_size_ * 0.5f;
        for (f32 t = t_enter; t <= t_exit; t += step) {
            f32 px = ox + dx * t;
            f32 py = oy + dy * t;
            f32 pz = oz + dz * t;
            // Bounds check
            if (px < n.x0 || px > n.x1 || pz < n.z0 || pz > n.z1) continue;
            u32 gx = std::min(static_cast<u32>(px / cell_size_), hmap_w_ - 1);
            u32 gz = std::min(static_cast<u32>(pz / cell_size_), hmap_h_ - 1);
            f32 h = heights_[gz * hmap_w_ + gx];
            if (py <= h && t < best_t) {
                best_t = t;
                hx = px; hy = h; hz = pz;
                return true;
            }
        }
        return false;
    }

    // Sort children by distance, traverse closest first
    struct ChildDist { i32 idx; f32 t; };
    ChildDist sorted[4];
    int count = 0;
    for (int c = 0; c < 4; ++c) {
        if (n.children[c] < 0) continue;
        f32 te, tx;
        const auto& cn = nodes_[n.children[c]];
        if (ray_aabb(ox, oy, oz, dx, dy, dz,
                     cn.x0, cn.min_h, cn.z0, cn.x1, cn.max_h, cn.z1,
                     te, tx)) {
            sorted[count++] = {n.children[c], te};
        }
    }
    std::sort(sorted, sorted + count,
              [](const ChildDist& a, const ChildDist& b) { return a.t < b.t; });

    bool hit = false;
    for (int i = 0; i < count; ++i) {
        if (sorted[i].t >= best_t) break;
        if (raycast_node(sorted[i].idx, ox, oy, oz, dx, dy, dz,
                         best_t, hx, hy, hz)) {
            hit = true;
        }
    }
    return hit;
}
```

```cpp
#include <spdlog/spdlog.h>
#include <string>

extern "C" {
#include <lua.h>
#include <lauxlib.h>

namespace osc::lua {

void deep_copy_value(lua_State* src, int idx, lua_State* dst) {
    int abs_idx = (idx > 0) ? idx : lua_gettop(src) + idx + 1;
    int t = lua_type(src, abs_idx);

    switch (t) {
    case LUA_TNIL:
        lua_pushnil(dst);
        break;
    case LUA_TBOOLEAN:
        lua_pushboolean(dst, lua_toboolean(src, abs_idx));
        break;
    case LUA_TNUMBER:
        lua_pushnumber(dst, lua_tonumber(src, abs_idx));
        break;
    case LUA_TSTRING: {
        size_t len;
        const char* s = lua_tolstring(src, abs_idx, &len);
        lua_pushlstring(dst, s, len);
        break;
    }
    case LUA_TTABLE:
        deep_copy_table(src, abs_idx, dst);
        break;
    default:
        // Functions, userdata, threads — push nil
        lua_pushnil(dst);
        break;
    }
}

void deep_copy_table(lua_State* src, int src_idx, lua_State* dst) {
    int abs = (src_idx > 0) ? src_idx : lua_gettop(src) + src_idx + 1;
    lua_newtable(dst);
    lua_pushnil(src);
    while (lua_next(src, abs) != 0) {
        // key at -2, value at -1
        deep_copy_value(src, -2, dst); // copy key
        deep_copy_value(src, -1, dst); // copy value
        lua_rawset(dst, -3);
        lua_pop(src, 1); // pop value, keep key
    }
}

void dispatch_sim_callback(lua_State* ui_L, lua_State* sim_L) {
    // Read Func name from ui_L stack top table
    if (!lua_istable(ui_L, -1)) {
        spdlog::warn("SimCallback: argument is not a table");
        return;
    }

    lua_pushstring(ui_L, "Func");
    lua_rawget(ui_L, -2);
    if (!lua_isstring(ui_L, -1)) {
        spdlog::warn("SimCallback: missing Func field");
        lua_pop(ui_L, 1);
        return;
    }
    std::string func_name = lua_tostring(ui_L, -1);
    lua_pop(ui_L, 1);

    // Deep-copy Args table to sim_L
    lua_pushstring(ui_L, "Args");
    lua_rawget(ui_L, -2);
    bool has_args = lua_istable(ui_L, -1);

    // On sim_L: look up SimCallbacks[func_name]
    lua_pushstring(sim_L, "SimCallbacks");
    lua_rawget(sim_L, LUA_GLOBALSINDEX);
    if (!lua_istable(sim_L, -1)) {
        spdlog::warn("SimCallback: SimCallbacks table not found");
        lua_pop(sim_L, 1);
        lua_pop(ui_L, 1);
        return;
    }

    lua_pushstring(sim_L, func_name.c_str());
    lua_rawget(sim_L, -2);
    if (!lua_isfunction(sim_L, -1)) {
        spdlog::warn("SimCallback: handler '{}' not found", func_name);
        lua_pop(sim_L, 2); // pop nil + SimCallbacks
        lua_pop(ui_L, 1);
        return;
    }

    // Push deep-copied args table as argument
    if (has_args) {
        deep_copy_table(ui_L, -1, sim_L);
    } else {
        lua_newtable(sim_L);
    }
    lua_pop(ui_L, 1); // pop Args from ui_L

    // Call handler(args) on sim_L
    if (lua_pcall(sim_L, 1, 0, 0) != 0) {
        spdlog::warn("SimCallback '{}' error: {}", func_name,
                     lua_tostring(sim_L, -1));
        lua_pop(sim_L, 1);
    }

    lua_pop(sim_L, 1); // pop SimCallbacks table
}

} // namespace osc::lua
```

```cpp
/// Check if a structure can be placed at (x, z) with given footprint.
/// Returns true if placement is valid (no overlap with existing structures,
/// terrain slope within limits).
bool SimState::can_build_at(f32 x, f32 z, f32 foot_x, f32 foot_z) const {
    if (!pathfinding_grid_) return true;
    // Check all cells covered by footprint for Impassable flag
    // (existing build ghost validity check uses Impassable only, not Obstacle)
    i32 cx0 = static_cast<i32>((x - foot_x * 0.5f));
    i32 cz0 = static_cast<i32>((z - foot_z * 0.5f));
    i32 cx1 = static_cast<i32>((x + foot_x * 0.5f));
    i32 cz1 = static_cast<i32>((z + foot_z * 0.5f));
    for (i32 cz = cz0; cz < cz1; ++cz) {
        for (i32 cx = cx0; cx < cx1; ++cx) {
            if (pathfinding_grid_->is_impassable(cx, cz)) return false;
        }
    }
    return true;
}
```
