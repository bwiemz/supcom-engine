In `src/renderer/mesh_cache.cpp`, implement `load_lod_set()`:

```cpp
bool MeshCache::load_lod_set(const std::string& bp_id, lua_State* L) {
    if (device_ == VK_NULL_HANDLE || !store_ || !L) return false;
    if (lod_cache_.count(bp_id) || failed_.count(bp_id)) return lod_cache_.count(bp_id) > 0;

    std::string mesh_bp_id = resolve_mesh_bp_id(bp_id, L);
    if (mesh_bp_id.empty()) {
        failed_.insert(bp_id);
        return false;
    }

    LODSet set;

    // Try loading LODs[1], LODs[2], etc. FA typically has 1-4 LOD levels.
    for (i32 lod_idx = 1; lod_idx <= 4; lod_idx++) {
        std::string mesh_path = resolve_mesh_path_for_lod(mesh_bp_id, lod_idx, L);
        if (mesh_path.empty()) break; // no more LOD levels

        auto file_data = vfs_->read_file(mesh_path);
        if (!file_data) continue; // skip this LOD if file missing

        auto mesh = sim::parse_scm_mesh(*file_data);
        if (!mesh || mesh->vertices.empty() || mesh->indices.empty()) continue;

        auto vert_buf = upload_buffer(device_, allocator_, cmd_pool_, queue_,
            mesh->vertices.data(), mesh->vertices.size() * sizeof(sim::SCMMesh::Vertex),
            VK_BUFFER_USAGE_VERTEX_BUFFER_BIT);
        if (!vert_buf.buffer) continue;

        auto idx_buf = upload_buffer(device_, allocator_, cmd_pool_, queue_,
            mesh->indices.data(), mesh->indices.size() * sizeof(u32),
            VK_BUFFER_USAGE_INDEX_BUFFER_BIT);
        if (!idx_buf.buffer) {
            vmaDestroyBuffer(allocator_, vert_buf.buffer, vert_buf.allocation);
            continue;
        }

        f32 scale = resolve_uniform_scale(bp_id, L);
        f32 cutoff = read_lod_cutoff(mesh_bp_id, lod_idx, L);

        LODEntry entry;
        entry.mesh.vertex_buf = vert_buf;
        entry.mesh.index_buf = idx_buf;
        entry.mesh.index_count = static_cast<u32>(mesh->indices.size());
        entry.mesh.uniform_scale = scale;
        entry.mesh.texture_path = resolve_albedo_path_for_lod(mesh_bp_id, lod_idx, L);
        entry.mesh.specteam_path = resolve_specteam_path_for_lod(mesh_bp_id, lod_idx, L);
        entry.mesh.normal_path = resolve_normal_path_for_lod(mesh_bp_id, lod_idx, L);
        entry.cutoff = cutoff;

        set.lods.push_back(std::move(entry));
    }

    if (set.lods.empty()) {
        // Fall back to existing single-LOD loading (resolve_mesh_path uses LODs[1])
        // This handles the common case where ExtractMeshBlueprint overwrites LODs[1]
        // with only ShaderName+LODCutoff and the mesh path must be derived by convention.
        std::string mesh_path = resolve_mesh_path(bp_id, L);
        if (mesh_path.empty()) { failed_.insert(bp_id); return false; }

        auto file_data = vfs_->read_file(mesh_path);
        if (!file_data) { failed_.insert(bp_id); return false; }

        auto mesh = sim::parse_scm_mesh(*file_data);
        if (!mesh || mesh->vertices.empty() || mesh->indices.empty()) {
            failed_.insert(bp_id); return false;
        }

        auto vert_buf = upload_buffer(device_, allocator_, cmd_pool_, queue_,
            mesh->vertices.data(), mesh->vertices.size() * sizeof(sim::SCMMesh::Vertex),
            VK_BUFFER_USAGE_VERTEX_BUFFER_BIT);
        if (!vert_buf.buffer) { failed_.insert(bp_id); return false; }

        auto idx_buf = upload_buffer(device_, allocator_, cmd_pool_, queue_,
            mesh->indices.data(), mesh->indices.size() * sizeof(u32),
            VK_BUFFER_USAGE_INDEX_BUFFER_BIT);
        if (!idx_buf.buffer) {
            vmaDestroyBuffer(allocator_, vert_buf.buffer, vert_buf.allocation);
            failed_.insert(bp_id); return false;
        }

        LODEntry entry;
        entry.mesh.vertex_buf = vert_buf;
        entry.mesh.index_buf = idx_buf;
        entry.mesh.index_count = static_cast<u32>(mesh->indices.size());
        entry.mesh.uniform_scale = resolve_uniform_scale(bp_id, L);
        entry.mesh.texture_path = resolve_albedo_path(bp_id, L);
        entry.mesh.specteam_path = resolve_specteam_path(bp_id, L);
        entry.mesh.normal_path = resolve_normal_path(bp_id, L);
        entry.cutoff = 0.0f; // no cutoff — use at all distances
        set.lods.push_back(std::move(entry));
    }

    // Sort by cutoff ascending (highest detail = smallest cutoff first)
    std::sort(set.lods.begin(), set.lods.end(),
              [](const LODEntry& a, const LODEntry& b) { return a.cutoff < b.cutoff; });

    spdlog::debug("MeshCache: loaded {} LOD levels for '{}'", set.lods.size(), bp_id);
    lod_cache_[bp_id] = std::move(set);
    return true;
}
```

```cpp
f32 MeshCache::read_lod_cutoff(const std::string& mesh_bp_id, i32 lod_index, lua_State* L) {
    lua_pushstring(L, "__blueprints");
    lua_rawget(L, LUA_GLOBALSINDEX);
    if (!lua_istable(L, -1)) { lua_pop(L, 1); return 0.0f; }
    int bps = lua_gettop(L);

    lua_pushstring(L, mesh_bp_id.c_str());
    lua_rawget(L, bps);
    if (!lua_istable(L, -1)) { lua_pop(L, 2); return 0.0f; }
    int mbp = lua_gettop(L);

    lua_pushstring(L, "LODs");
    lua_rawget(L, mbp);
    if (!lua_istable(L, -1)) { lua_pop(L, 3); return 0.0f; }
    int lods = lua_gettop(L);

    lua_rawgeti(L, lods, lod_index);
    if (!lua_istable(L, -1)) { lua_pop(L, 4); return 0.0f; }
    int lod_table = lua_gettop(L);

    lua_pushstring(L, "LODCutoff");
    lua_rawget(L, lod_table);
    f32 cutoff = 0.0f;
    if (lua_isnumber(L, -1)) {
        cutoff = static_cast<f32>(lua_tonumber(L, -1));
    }
    lua_pop(L, 5); // LODCutoff + lod_table + lods + mbp + bps
    return cutoff;
}
```

```cpp
std::string MeshCache::resolve_mesh_path_for_lod(const std::string& mesh_bp_id,
                                                   i32 lod_index, lua_State* L) {
    lua_pushstring(L, "__blueprints");
    lua_rawget(L, LUA_GLOBALSINDEX);
    if (!lua_istable(L, -1)) { lua_pop(L, 1); return {}; }
    int bps = lua_gettop(L);

    lua_pushstring(L, mesh_bp_id.c_str());
    lua_rawget(L, bps);
    if (!lua_istable(L, -1)) { lua_pop(L, 2); return {}; }
    int mbp = lua_gettop(L);

    lua_pushstring(L, "LODs");
    lua_rawget(L, mbp);
    if (!lua_istable(L, -1)) { lua_pop(L, 3); return {}; }
    int lods = lua_gettop(L);

    lua_rawgeti(L, lods, lod_index);
    if (!lua_istable(L, -1)) { lua_pop(L, 4); return {}; }
    int lod_table = lua_gettop(L);

    lua_pushstring(L, "MeshName");
    lua_rawget(L, lod_table);
    std::string result;
    if (lua_isstring(L, -1)) {
        result = lua_tostring(L, -1);
        if (!result.empty() && result[0] != '/') {
            auto slash = mesh_bp_id.rfind('/');
            if (slash != std::string::npos)
                result = mesh_bp_id.substr(0, slash + 1) + result;
        }
    }
    lua_pop(L, 5); // MeshName + lod_table + lods + mbp + bps
    return result;
}
```

**Step 3: Implement get_lod() and update get()**

```cpp
const GPUMesh* MeshCache::get_lod(const std::string& blueprint_id,
                                    f32 camera_distance, lua_State* L) {
    auto it = lod_cache_.find(blueprint_id);
    if (it == lod_cache_.end()) {
        if (failed_.count(blueprint_id)) return nullptr;
        if (!load_lod_set(blueprint_id, L)) return nullptr;
        it = lod_cache_.find(blueprint_id);
        if (it == lod_cache_.end()) return nullptr;
    }

    const auto& lods = it->second.lods;
    if (lods.empty()) return nullptr;

    // Walk LODs from highest detail (index 0) to lowest.
    // Use the first LOD whose cutoff >= camera_distance (or cutoff == 0 meaning "any distance").
    // If camera_distance exceeds all cutoffs, use the last (lowest detail) LOD.
    for (const auto& entry : lods) {
        if (entry.cutoff <= 0.0f || camera_distance <= entry.cutoff) {
            return &entry.mesh;
        }
    }
    return &lods.back().mesh; // beyond all cutoffs — use lowest detail
}

const LODSet* MeshCache::get_lod_set(const std::string& blueprint_id) const {
    auto it = lod_cache_.find(blueprint_id);
    return (it != lod_cache_.end()) ? &it->second : nullptr;
}

// Update existing get() to delegate to get_lod with distance 0 (highest detail):
const GPUMesh* MeshCache::get(const std::string& blueprint_id, lua_State* L) {
    return get_lod(blueprint_id, 0.0f, L);
}
```

```cpp
    if (!dying_) return;
```

**Step 2: Write shader sources in shader_utils.cpp**

Add the following shader source strings. All bloom shaders use a fullscreen triangle generated from `gl_VertexIndex` (no vertex buffer needed).

**Fullscreen triangle vertex shader** (shared by all bloom passes):
```glsl
// bloom_bright_vert
#version 450
layout(location = 0) out vec2 fragUV;

void main() {
    // Fullscreen triangle: 3 vertices cover the entire screen
    fragUV = vec2((gl_VertexIndex << 1) & 2, gl_VertexIndex & 2);
    gl_Position = vec4(fragUV * 2.0 - 1.0, 0.0, 1.0);
    fragUV.y = 1.0 - fragUV.y; // flip Y for Vulkan
}
```

**Brightness extraction fragment shader:**
```glsl
// bloom_bright_frag
#version 450
layout(location = 0) in vec2 fragUV;
layout(location = 0) out vec4 outColor;

layout(set = 0, binding = 0) uniform sampler2D sceneTex;

layout(push_constant) uniform PC {
    float threshold;
    float intensity;
} pc;

void main() {
    vec3 color = texture(sceneTex, fragUV).rgb;
    float brightness = dot(color, vec3(0.2126, 0.7152, 0.0722));
    vec3 bright = max(color - vec3(pc.threshold), vec3(0.0));
    outColor = vec4(bright * pc.intensity, 1.0);
}
```

**Separable Gaussian blur fragment shader:**
```glsl
// bloom_blur_frag
#version 450
layout(location = 0) in vec2 fragUV;
layout(location = 0) out vec4 outColor;

layout(set = 0, binding = 0) uniform sampler2D inputTex;

layout(push_constant) uniform PC {
    vec2 direction;  // (1/w, 0) for horizontal, (0, 1/h) for vertical
} pc;

void main() {
    // 9-tap Gaussian kernel (sigma ~= 2.5)
    const float weights[5] = float[](0.2270270270, 0.1945945946, 0.1216216216, 0.0540540541, 0.0162162162);

    vec3 result = texture(inputTex, fragUV).rgb * weights[0];
    for (int i = 1; i < 5; i++) {
        vec2 offset = pc.direction * float(i);
        result += texture(inputTex, fragUV + offset).rgb * weights[i];
        result += texture(inputTex, fragUV - offset).rgb * weights[i];
    }
    outColor = vec4(result, 1.0);
}
```

**Composite fragment shader:**
```glsl
// bloom_composite_frag
#version 450
layout(location = 0) in vec2 fragUV;
layout(location = 0) out vec4 outColor;

layout(set = 0, binding = 0) uniform sampler2D sceneTex;
layout(set = 1, binding = 0) uniform sampler2D bloomTex;

layout(push_constant) uniform PC {
    float bloomStrength;
} pc;

void main() {
    vec3 scene = texture(sceneTex, fragUV).rgb;
    vec3 bloom = texture(bloomTex, fragUV).rgb;
    outColor = vec4(scene + bloom * pc.bloomStrength, 1.0);
}
```

```cpp
        bloom_blur_v_ds_ = sets[3];

            VkDescriptorImageInfo img_info{};
            img_info.sampler = texture_sampler_;
            img_info.imageView = view;

        write_ds(bloom_blur_h_ds_, bloom_blur_h_image_.view);
        write_ds(bloom_blur_v_ds_, bloom_blur_v_image_.view);
    }
```

### Task 9: Create bloom pipelines and record bloom passes

**Files:**
- Modify: `src/renderer/renderer.cpp`

**Step 1: Implement create_bloom_pipelines()**

Create 3 pipelines using `PipelineBuilder`:
1. **Bright pass**: Uses `bloom_bright_vert` + `bloom_bright_frag`, push constants = `{threshold, intensity}` (8 bytes), descriptor set 0 = scene texture.
2. **Blur pass**: Uses `bloom_bright_vert` + `bloom_blur_frag`, push constants = `{direction.x, direction.y}` (8 bytes), descriptor set 0 = input texture.
3. **Composite pass**: Uses `bloom_bright_vert` + `bloom_composite_frag`, push constants = `{bloomStrength}` (4 bytes), descriptor sets 0 = scene, 1 = bloom result.

All use no vertex input (fullscreen triangle from gl_VertexIndex), no depth test, no depth write, no blending. Render to bloom_render_pass_ (half-res for bright/blur) or the swapchain render pass (for composite).

Key: The composite pipeline renders to the **swapchain** framebuffer (at full resolution), so it uses `render_pass_` (the existing main render pass), not `bloom_render_pass_`.

**Step 2: Modify render() to use bloom pipeline**

When bloom is enabled, the render flow changes:

1. **Main scene** → render to `scene_color_image_` (via `scene_render_pass_` + `scene_framebuffer_`) instead of swapchain.
2. **Bright pass** → render to `bloom_bright_image_` at half-res, sampling `scene_color_image_`.
3. **Blur horizontal** → render to `bloom_blur_h_image_`, sampling `bloom_bright_image_`.
4. **Blur vertical** → render to `bloom_blur_v_image_`, sampling `bloom_blur_h_image_`.
5. **Composite** → render to swapchain, sampling `scene_color_image_` (set 0) and `bloom_blur_v_image_` (set 1).

Between each pass, insert `VkImageMemoryBarrier` to transition image layouts from `COLOR_ATTACHMENT_OPTIMAL` to `SHADER_READ_ONLY_OPTIMAL`.

When bloom is disabled, use the existing direct-to-swapchain path unchanged.

```cpp
    vkCmdSetScissor(cmd_buf_[fi], 0, 1, &scissor);

    f32 strength = bloom_strength_;
    vkCmdPushConstants(cmd_buf_[fi], bloom_composite_layout_,
                        VK_SHADER_STAGE_FRAGMENT_BIT, 0, sizeof(strength), &strength);

    vkCmdBindDescriptorSets(cmd_buf_[fi], VK_PIPELINE_BIND_POINT_GRAPHICS,
                             bloom_composite_layout_, 0, 1, &scene_ds_, 0, nullptr);
    vkCmdBindDescriptorSets(cmd_buf_[fi], VK_PIPELINE_BIND_POINT_GRAPHICS,
                             bloom_composite_layout_, 1, 1, &bloom_blur_v_ds_, 0, nullptr);

    vkCmdDraw(cmd_buf_[fi], 3, 1, 0, 0);

    // Then draw UI on top (UI renders after composite, directly to swapchain)
    // ... (strategic icons, overlays, minimap, HUD, selection info, UI, profile overlay)
}
```
