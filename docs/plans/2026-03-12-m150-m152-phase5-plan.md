FA registers key maps via `IN_AddKeyMapTable(keymap)` where `keymap` is a table of `{keyName = {action="func_name"}}` entries. Multiple key map tables can be stacked (e.g., game mode, build mode, command mode).

- [ ] **Step 1: Write keymap header**

```cpp
#pragma once

#include <string>
#include <unordered_map>
#include <vector>

struct lua_State;

namespace osc::ui {

/// A single key mapping: key name → Lua action function reference.
struct KeyBinding {
    std::string key_name;      // e.g., "A", "Shift-G", "Ctrl-1"
    int action_ref = -2;       // Lua registry ref to the action function
    std::string action_name;   // Debug: action name string
};

/// A set of key bindings (one IN_AddKeyMapTable call).
/// Identified by Lua registry ref to the original table for removal.
struct KeyMapTable {
    int table_ref = -2;  // Lua registry ref to the original table (for IN_RemoveKeyMapTable)
    std::vector<KeyBinding> bindings;
};

/// Stack of key map tables. Later-added tables take priority (searched last-to-first).
class KeyMapRegistry {
public:
    /// Add a key map table from Lua. Parses the table at stack position `table_idx`.
    /// Stores a registry ref to the table for later removal via remove_by_ref().
    void add(lua_State* L, int table_idx);

    /// Remove a key map table by its original Lua table reference.
    /// FA calls IN_RemoveKeyMapTable(table) passing the same table that was added.
    void remove_by_ref(lua_State* L, int table_idx);

    /// Look up an action for a key press. Returns the Lua registry ref of
    /// the action function, or LUA_NOREF if no binding matches.
    /// Searches tables in reverse order (last added = highest priority).
    int find_action(const std::string& key_name) const;

    /// Fire the action for a key press. Returns true if a binding was found.
    bool dispatch(lua_State* L, const std::string& key_name);

    /// Clear all tables (e.g., on state transition).
    void clear(lua_State* L);

    /// Convert a GLFW key code + modifiers to an FA-style key name string.
    /// e.g., GLFW_KEY_A + Shift → "Shift-A", GLFW_KEY_1 + Ctrl → "Ctrl-1"
    static std::string glfw_to_key_name(int glfw_key, int mods);

private:
    std::vector<KeyMapTable> tables_;
};

} // namespace osc::ui
```

#include "ui/keymap.hpp"
#include <spdlog/spdlog.h>
#include <GLFW/glfw3.h>

extern "C" {
#include <lua.h>
#include <lauxlib.h>
}

namespace osc::ui {

void KeyMapRegistry::add(lua_State* L, int table_idx) {
    KeyMapTable kmt;

    // Store a ref to the original table so we can identify it during remove
    lua_pushvalue(L, table_idx);
    kmt.table_ref = luaL_ref(L, LUA_REGISTRYINDEX);

    // Iterate the Lua table: keys are key names, values are action tables or functions
    lua_pushnil(L);
    while (lua_next(L, table_idx) != 0) {
        if (lua_type(L, -2) == LUA_TSTRING) {
            std::string key = lua_tostring(L, -2);
            KeyBinding kb;
            kb.key_name = key;

            if (lua_isfunction(L, -1)) {
                lua_pushvalue(L, -1);
                kb.action_ref = luaL_ref(L, LUA_REGISTRYINDEX);
                kb.action_name = key;
            } else if (lua_istable(L, -1)) {
                // Table format: {action = function}
                lua_pushstring(L, "action");
                lua_rawget(L, -2);
                if (lua_isfunction(L, -1)) {
                    kb.action_ref = luaL_ref(L, LUA_REGISTRYINDEX);
                } else {
                    lua_pop(L, 1);
                }
            }

            if (kb.action_ref >= 0) {
                kmt.bindings.push_back(std::move(kb));
            }
        }
        lua_pop(L, 1); // pop value, keep key for next
    }

    spdlog::debug("KeyMap added: {} bindings", kmt.bindings.size());
    tables_.push_back(std::move(kmt));
}

void KeyMapRegistry::remove_by_ref(lua_State* L, int table_idx) {
    // FA calls IN_RemoveKeyMapTable(keymap) with the same table that was added.
    // Compare by Lua table pointer identity.
    const void* target_ptr = lua_topointer(L, table_idx);
    for (auto it = tables_.begin(); it != tables_.end(); ++it) {
        // Retrieve the stored table ref and compare pointers
        lua_rawgeti(L, LUA_REGISTRYINDEX, it->table_ref);
        const void* stored_ptr = lua_topointer(L, -1);
        lua_pop(L, 1);
        if (stored_ptr == target_ptr) {
            for (auto& kb : it->bindings) {
                luaL_unref(L, LUA_REGISTRYINDEX, kb.action_ref);
            }
            luaL_unref(L, LUA_REGISTRYINDEX, it->table_ref);
            tables_.erase(it);
            return;
        }
    }
    spdlog::debug("KeyMap remove: table not found");
}

int KeyMapRegistry::find_action(const std::string& key_name) const {
    // Search last-to-first (highest priority first)
    for (auto it = tables_.rbegin(); it != tables_.rend(); ++it) {
        for (const auto& kb : it->bindings) {
            if (kb.key_name == key_name) return kb.action_ref;
        }
    }
    return -2; // LUA_NOREF
}

bool KeyMapRegistry::dispatch(lua_State* L, const std::string& key_name) {
    int ref = find_action(key_name);
    if (ref < 0) return false;

    lua_rawgeti(L, LUA_REGISTRYINDEX, ref);
    if (lua_isfunction(L, -1)) {
        if (lua_pcall(L, 0, 0, 0) != 0) {
            spdlog::warn("KeyMap '{}' error: {}", key_name, lua_tostring(L, -1));
            lua_pop(L, 1);
        }
        return true;
    }
    lua_pop(L, 1);
    return false;
}

void KeyMapRegistry::clear(lua_State* L) {
    for (auto& t : tables_) {
        for (auto& kb : t.bindings) {
            luaL_unref(L, LUA_REGISTRYINDEX, kb.action_ref);
        }
        luaL_unref(L, LUA_REGISTRYINDEX, t.table_ref);
    }
    tables_.clear();
}

std::string KeyMapRegistry::glfw_to_key_name(int glfw_key, int mods) {
    std::string result;
    if (mods & GLFW_MOD_CONTROL) result += "Ctrl-";
    if (mods & GLFW_MOD_SHIFT) result += "Shift-";
    if (mods & GLFW_MOD_ALT) result += "Alt-";

    // Map common keys
    if (glfw_key >= GLFW_KEY_A && glfw_key <= GLFW_KEY_Z) {
        result += static_cast<char>('A' + (glfw_key - GLFW_KEY_A));
    } else if (glfw_key >= GLFW_KEY_0 && glfw_key <= GLFW_KEY_9) {
        result += static_cast<char>('0' + (glfw_key - GLFW_KEY_0));
    } else if (glfw_key >= GLFW_KEY_F1 && glfw_key <= GLFW_KEY_F12) {
        result += "F" + std::to_string(1 + (glfw_key - GLFW_KEY_F1));
    } else {
        switch (glfw_key) {
        case GLFW_KEY_ESCAPE: result += "Escape"; break;
        case GLFW_KEY_SPACE:  result += "Space"; break;
        case GLFW_KEY_ENTER:  result += "Enter"; break;
        case GLFW_KEY_TAB:    result += "Tab"; break;
        case GLFW_KEY_DELETE: result += "Delete"; break;
        case GLFW_KEY_PAUSE:  result += "Pause"; break;
        default: result += "Key" + std::to_string(glfw_key); break;
        }
    }
    return result;
}

} // namespace osc::ui

/// IN_AddKeyMapTable(keymap) — register a key map table
static int l_IN_AddKeyMapTable(lua_State* L) {
    auto* reg = get_keymap_registry(L);
    if (!reg || !lua_istable(L, 1)) return 0;

    reg->add(L, 1);
    return 0;
}

/// IN_RemoveKeyMapTable(keymap) — unregister a key map table by table reference.
/// FA passes the same table that was registered — we match by pointer identity.
static int l_IN_RemoveKeyMapTable(lua_State* L) {
    auto* reg = get_keymap_registry(L);
    if (!reg || !lua_istable(L, 1)) return 0;

    reg->remove_by_ref(L, 1);
    return 0;
}

/// IsKeyDown(keyName) → boolean
/// Queries current GLFW key state for modifier checks.
static int l_IsKeyDown(lua_State* L) {
    const char* key_name = luaL_checkstring(L, 1);

    // Get GLFW window from registry
    lua_pushstring(L, "__osc_glfw_window");
    lua_rawget(L, LUA_REGISTRYINDEX);
    auto* window = static_cast<GLFWwindow*>(lua_touserdata(L, -1));
    lua_pop(L, 1);

    if (!window) { lua_pushboolean(L, 0); return 1; }

    int glfw_key = 0;
    std::string k(key_name);
    if (k == "Shift") glfw_key = GLFW_KEY_LEFT_SHIFT;
    else if (k == "Control" || k == "Ctrl") glfw_key = GLFW_KEY_LEFT_CONTROL;
    else if (k == "Alt") glfw_key = GLFW_KEY_LEFT_ALT;
    else if (k.size() == 1 && k[0] >= 'A' && k[0] <= 'Z')
        glfw_key = GLFW_KEY_A + (k[0] - 'A');

    bool down = glfw_key > 0 && glfwGetKey(window, glfw_key) == GLFW_PRESS;
    // Also check right-side modifier keys
    if (!down && k == "Shift") down = glfwGetKey(window, GLFW_KEY_RIGHT_SHIFT) == GLFW_PRESS;
    if (!down && (k == "Control" || k == "Ctrl"))
        down = glfwGetKey(window, GLFW_KEY_RIGHT_CONTROL) == GLFW_PRESS;
    if (!down && k == "Alt") down = glfwGetKey(window, GLFW_KEY_RIGHT_ALT) == GLFW_PRESS;

    lua_pushboolean(L, down ? 1 : 0);
    return 1;
}

/// UIZoomTo(units, duration) — animate camera to center on units.
/// units is a Lua array of unit objects.
static int l_UIZoomTo(lua_State* L) {
    if (!lua_istable(L, 1)) return 0;
    f32 duration = static_cast<f32>(luaL_optnumber(L, 2, 0.5));

    // Compute centroid of all units
    auto* sim = get_sim(L);
    if (!sim) return 0;

    f32 sum_x = 0, sum_z = 0;
    int count = 0;
    int n = luaL_getn(L, 1);
    for (int i = 1; i <= n; ++i) {
        lua_rawgeti(L, 1, i);
        // Extract _c_object
        if (lua_istable(L, -1)) {
            lua_pushstring(L, "_c_object");
            lua_rawget(L, -2);
            auto* unit = static_cast<osc::sim::Unit*>(lua_touserdata(L, -1));
            lua_pop(L, 1);
            if (unit) {
                auto pos = unit->position();
                sum_x += pos.x;
                sum_z += pos.z;
                ++count;
            }
        }
        lua_pop(L, 1);
    }

    if (count > 0) {
        // Set camera target to centroid
        // Access camera via registry
        lua_pushstring(L, "__osc_camera");
        lua_rawget(L, LUA_REGISTRYINDEX);
        auto* cam = static_cast<osc::renderer::Camera*>(lua_touserdata(L, -1));
        lua_pop(L, 1);
        if (cam) {
            cam->set_target(sum_x / count, sum_z / count);
        }
    }

    return 0;
}

/// Calls the registered chat function with sender = "System".
static int l_SendSystemMessage(lua_State* L) {

        if (lua_pcall(L, 1, 0, 0) != 0) {
            spdlog::warn("SendSystemMessage error: {}", lua_tostring(L, -1));
            lua_pop(L, 1);
        }

/// StartLoadingDialog(title) — show loading screen
static int l_StartLoadingDialog(lua_State* L) {
    const char* title = luaL_optstring(L, 1, "Loading...");
    lua_pushstring(L, "__osc_loading_title");
    lua_pushstring(L, title);
    lua_rawset(L, LUA_REGISTRYINDEX);
    lua_pushstring(L, "__osc_loading_progress");
    lua_pushnumber(L, 0);
    lua_rawset(L, LUA_REGISTRYINDEX);
    spdlog::info("Loading: {}", title);
    return 0;
}

/// UpdateLoadingDialog(progress) — update progress (0.0 to 1.0)
static int l_UpdateLoadingDialog(lua_State* L) {
    f32 progress = static_cast<f32>(luaL_checknumber(L, 1));
    lua_pushstring(L, "__osc_loading_progress");
    lua_pushnumber(L, progress);
    lua_rawset(L, LUA_REGISTRYINDEX);
    spdlog::debug("Loading progress: {:.0f}%", progress * 100);
    return 0;
}

/// StopLoadingDialog() — dismiss loading screen
static int l_StopLoadingDialog(lua_State* L) {
    lua_pushstring(L, "__osc_loading_title");
    lua_pushnil(L);
    lua_rawset(L, LUA_REGISTRYINDEX);
    spdlog::info("Loading complete");
    return 0;
}
